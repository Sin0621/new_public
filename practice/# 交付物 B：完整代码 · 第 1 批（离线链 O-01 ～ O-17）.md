## 5. `app/core/logging.py`  
  
```python  
# 职责：初始化控制台与轮转文件日志，统一提供 get_logger。  
# 实现步骤：O-04  
# 被谁调用：app/main.py；各模块用 get_logger(__name__)  
# ── 【O-04】初始化日志 ──  
"""日志设施。"""  
import logging  
import sys  
from logging.handlers import RotatingFileHandler  
from pathlib import Path  
  
from app.config import settings  
  
LOG_FORMAT = "%(asctime)s | %(levelname)-8s | %(name)s | %(message)s"  
DATE_FORMAT = "%Y-%m-%d %H:%M:%S"  
  
  
def setup_logging() -> None:  
    """配置根日志器；重复调用不会重复添加处理器。"""  
    root = logging.getLogger()  
    if root.handlers:  
        return  
  
    root.setLevel(settings.LOG_LEVEL.upper())  
    formatter = logging.Formatter(LOG_FORMAT, datefmt=DATE_FORMAT)  
  
    console = logging.StreamHandler(sys.stdout)  
    console.setFormatter(formatter)  
    root.addHandler(console)  
  
    log_dir = Path(settings.LOG_DIR)  
    log_dir.mkdir(parents=True, exist_ok=True)  
    file_handler = RotatingFileHandler(  
        filename=log_dir / "app.log",  
        maxBytes=10 * 1024 * 1024,  
        backupCount=5,  
        encoding="utf-8",  
    )  
    file_handler.setFormatter(formatter)  
    root.addHandler(file_handler)  
  
    logging.getLogger("httpx").setLevel(logging.WARNING)  
  
  
def get_logger(name: str) -> logging.Logger:  
    """按模块名取 logger。"""  
    return logging.getLogger(name)  
```

## 12. `app/services/loader.py`  
  
```python  
# 职责：把资料文件读成纯文本，失败时抛可读异常。  
# 实现步骤：O-10  
# 被谁调用：scripts/import_docs.py  
# ── 【O-10】读取资料文件 ──  
"""资料文件读取。"""  
from pathlib import Path  
  
from app.core.exceptions import AppError  
from app.core.logging import get_logger  
  
logger = get_logger(__name__)  
  
SUPPORTED_SUFFIXES = {".txt", ".md"}  
  
  
class FileParseError(AppError):  
    """文件读取或解析失败。"""  
  
    status_code = 400  
    code = "FILE_PARSE_ERROR"  
    message = "文件解析失败"  
  
    def __init__(self, path: str, reason: str):  
        """保留路径与原因，便于日志定位。"""  
        self.path = path  
        self.reason = reason  
        super().__init__(f"{path} 解析失败: {reason}")  
  
  
def load_text(path: str | Path) -> str:  
    """读取文本文件并返回全文。"""  
    p = Path(path)  
    if not p.exists():  
        raise FileParseError(str(p), "文件不存在")  
    if p.suffix.lower() not in SUPPORTED_SUFFIXES:  
        raise FileParseError(str(p), f"不支持的扩展名 {p.suffix}")  
  
    try:  
        text = p.read_text(encoding="utf-8")  
    except UnicodeDecodeError:  
        logger.warning("utf-8 解码失败，改用 gbk: %s", p)  
        try:  
            text = p.read_text(encoding="gbk")  
        except UnicodeDecodeError as exc:  
            raise FileParseError(str(p), "无法识别文件编码") from exc  
    except OSError as exc:  
        raise FileParseError(str(p), str(exc)) from exc  
  
    logger.info("读取完成 | file=%s | chars=%d", p.name, len(text))  
    return text  
```

## 13. `app/services/cleaner.py`  
  
```python  
# 职责：去掉 HTML 标签与代码块，压缩空白和空行。  
# 实现步骤：O-11  
# 被谁调用：scripts/import_docs.py、tests/test_pipeline.py  
# ── 【O-11】清洗文本 ──  
"""文本清洗。"""  
import re  
  
HTML_TAG = re.compile(r"<[^>]+>")  
CODE_BLOCK = re.compile(r"```.*?```", re.DOTALL)  
SPACES = re.compile(r"[ \t\u3000]+")  
BLANK_LINES = re.compile(r"\n{3,}")  
  
  
def clean_text(text: str) -> str:  
    """按固定顺序执行四步清洗。"""  
    text = HTML_TAG.sub("", text)  
    text = CODE_BLOCK.sub("", text)  
    text = SPACES.sub(" ", text)  
    text = BLANK_LINES.sub("\n\n", text)  
    return text.strip()  
```

## 14. `app/services/splitter.py`  
  
```python  
# 职责：按句末标点切句，再按长度装箱并保留重叠。  
# 实现步骤：O-12  
# 被谁调用：scripts/import_docs.py、tests/test_pipeline.py  
# ── 【O-12】切分文本 ──  
"""文本分块。"""  
import re  
  
SENTENCE = re.compile(r"[^。！？!?]+[。！？!?]?")  
  
  
def split_sentences(text: str) -> list[str]:  
    """把文本切成句子列表，过滤空白项。"""  
    return [s.strip() for s in SENTENCE.findall(text) if s.strip()]  
  
  
def split_text(text: str, size: int = 300, overlap: int = 50) -> list[dict]:  
    """按句子边界切块，返回带 index 与 char_count 的列表。"""  
    if not text:  
        return []  
    if overlap >= size:  
        raise ValueError("overlap 必须小于 size")  
  
    chunks: list[dict] = []  
    buf = ""  
  
    for sentence in split_sentences(text):  
        if len(buf) + len(sentence) <= size:  
            buf += sentence  
        else:  
            if buf:  
                chunks.append(  
                    {"index": len(chunks), "text": buf, "char_count": len(buf)}  
                )  
            buf = (buf[-overlap:] if overlap else "") + sentence  
  
    if buf:  
        chunks.append({"index": len(chunks), "text": buf, "char_count": len(buf)})  
  
    return chunks  
```

# 职责：文档的新增、按主键查询、列表与删除。  
# 实现步骤：O-14  
# 被谁调用：scripts/import_docs.py、app/api/v1/chat.py、tests/*  
# ── 【O-14】文档仓储 ──  
"""文档仓储。"""  
from sqlalchemy import select  
from sqlalchemy.orm import Session  
  
from app.models.document import Document  
  
  
def create_document(db: Session, filename: str, source_path: str) -> Document:  
    """新增文档并返回带主键的实例。"""  
    doc = Document(filename=filename, source_path=source_path)  
    db.add(doc)  
    db.commit()  
    db.refresh(doc)  
    return doc  
  
  
def get_document(db: Session, document_id: int) -> Document | None:  
    """按主键取文档，找不到返回 None。"""  
    return db.get(Document, document_id)  
  
  
def list_documents(db: Session) -> list[Document]:  
    """按 id 倒序返回全部文档。"""  
    stmt = select(Document).order_by(Document.id.desc())  
    return list(db.scalars(stmt).all())  
  
  
def delete_document(db: Session, document_id: int) -> bool:  
    """删除文档及其分块，返回是否删除成功。"""  
    doc = db.get(Document, document_id)  
    if doc is None:  
        return False  
    db.delete(doc)  
    db.commit()  
    return True

## 17. `app/services/ingest.py`  
  
```python  
# 职责：先删旧块再批量写入，保证重复导入不翻倍。  
# 实现步骤：O-15  
# 被谁调用：scripts/import_docs.py、tests/test_pipeline.py  
# ── 【O-15】批量入库 ──  
"""分块入库。"""  
from sqlalchemy import delete  
from sqlalchemy.exc import IntegrityError  
from sqlalchemy.orm import Session  
  
from app.core.exceptions import AppError  
from app.core.logging import get_logger  
from app.models.chunk import Chunk  
  
logger = get_logger(__name__)  
  
  
class IngestError(AppError):  
    """分块入库失败。"""  
  
    status_code = 409  
    code = "INGEST_CONFLICT"  
    message = "分块入库失败"  
  
    def __init__(self, document_id: int, reason: str):  
        """保留文档 id 与原因。"""  
        self.document_id = document_id  
        self.reason = reason  
        super().__init__(f"文档 {document_id} 入库失败: {reason}")  
  
  
def save_chunks(db: Session, document_id: int, chunks: list[dict]) -> int:  
    """覆盖式写入：先删该文档旧块，再批量插入。返回写入块数。"""  
    if not chunks:  
        return 0  
  
    try:  
        db.execute(delete(Chunk).where(Chunk.document_id == document_id))  
        db.add_all(  
            [  
                Chunk(  
                    document_id=document_id,  
                    index=item["index"],  
                    text=item["text"],  
                    char_count=item["char_count"],  
                )  
                for item in chunks  
            ]  
        )  
        db.commit()  
    except IntegrityError as exc:  
        db.rollback()  
        logger.error("写入冲突 | document_id=%s | %s", document_id, exc)  
        raise IngestError(document_id, "分块序号冲突") from exc  
  
    logger.info("写入完成 | document_id=%s | chunks=%d", document_id, len(chunks))  
    return len(chunks)  
```

from app.db.init_db import init_db  
from app.db.session import SessionLocal  
from app.repositories.document_repo import create_document  
from app.services.cleaner import clean_text  
from app.services.ingest import save_chunks  
from app.services.loader import load_text  
from app.services.splitter import split_text  
  
logger = get_logger(__name__)  
  
  
def main(path_str: str) -> None:  
    """读取一份资料文件并写入数据库。"""  
    setup_logging()  
    init_db()  
  
    path = Path(path_str)  
    text = clean_text(load_text(path))  
    chunks = split_text(text, size=300, overlap=50)  
  
    db = SessionLocal()  
    try:  
        doc = create_document(db, path.name, str(path))  
        count = save_chunks(db, doc.id, chunks)  
    finally:  
        db.close()  
  
    print(f"imported 1 document, {count} chunks")  
  
  
if __name__ == "__main__":  
    if len(sys.argv) != 2:  
        print("用法: python -m scripts.import_docs <文件路径>")  
        raise SystemExit(1)  
    main(sys.argv[1])
