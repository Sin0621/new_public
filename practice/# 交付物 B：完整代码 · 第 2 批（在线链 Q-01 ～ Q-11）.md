```python  
# 职责：定义消息、工具描述、模型决策三个运行时对象。  
# 实现步骤：Q-01  
# 被谁调用：app/agent/llm_client.py、app/agent/registry.py、app/agent/executor.py  
# ── 【Q-01】定义 Agent 协议模型 ──  
"""Agent 运行时协议。"""  
from typing import Literal  
  
from pydantic import BaseModel, ConfigDict, Field  
  
  
class ChatMessage(BaseModel):  
    """一条对话消息。"""  
  
    model_config = ConfigDict(extra="forbid")  
  
    role: Literal["system", "user", "assistant"]  
    content: str  
  
  
class ToolSpec(BaseModel):  
    """工具自描述，供模型选择与填参。"""  
  
    model_config = ConfigDict(extra="forbid")  
  
    name: str  
    description: str  
    parameters: dict = Field(default_factory=dict)  
  
  
class AgentDecision(BaseModel):  
    """模型对下一步的决策。"""  
  
    model_config = ConfigDict(extra="forbid")  
  
    thought: str = ""  
    action: Literal["tool", "final"]  
    tool: str | None = None  
    args: dict = Field(default_factory=dict)  
    answer: str | None = None  
```

## 3. `app/core/retry.py`  
  
```python  
# 职责：提供按次数与线性退避的重试装饰器。  
# 实现步骤：Q-02  
# 被谁调用：app/agent/llm_client.py  
# ── 【Q-02】重试装饰器 ──  
"""通用重试装饰器。"""  
import functools  
import time  
  
from app.core.logging import get_logger  
  
logger = get_logger(__name__)  
  
  
def retry(times: int = 3, backoff: float = 1.0, exceptions: tuple = (Exception,)):  
    """失败后按线性退避重试，最后一次仍失败则抛出。"""  
  
    def decorator(func):  
        @functools.wraps(func)  
        def wrapper(*args, **kwargs):  
            for attempt in range(1, times + 1):  
                try:  
                    return func(*args, **kwargs)  
                except exceptions as exc:  
                    if attempt == times:  
                        raise  
                    logger.warning(  
                        "第 %d 次失败，%.1fs 后重试: %s", attempt, backoff, exc  
                    )  
                    time.sleep(backoff * attempt)  
  
        return wrapper  
  
    return decorator  
```

## 4. `app/agent/llm_client.py`  
  
```python  
# 职责：调用 OpenAI 兼容的 chat/completions 接口并返回文本。  
# 实现步骤：Q-03  
# 被谁调用：scripts/ask.py、app/api/v1/chat.py、tests  
# ── 【Q-03】模型客户端 ──  
"""模型客户端。"""  
import httpx  
  
from app.agent.schemas import ChatMessage  
from app.config import settings  
from app.core.exceptions import UpstreamError  
from app.core.logging import get_logger  
from app.core.retry import retry  
  
logger = get_logger(__name__)  
  
  
class LLMClient:  
    """OpenAI 兼容接口的同步客户端。"""  
  
    def __init__(  
        self,  
        base_url: str | None = None,  
        api_key: str | None = None,  
        model: str | None = None,  
        timeout: float = 30.0,  
    ):  
        """允许测试时覆盖地址、密钥、模型与超时。"""  
        self.base_url = (base_url or settings.LLM_BASE_URL).rstrip("/")  
        self.api_key = api_key or settings.LLM_API_KEY  
        self.model = model or settings.LLM_MODEL  
        self.timeout = timeout  
  
    @retry(  
        times=3,  
        backoff=1.0,  
        exceptions=(httpx.TimeoutException, httpx.HTTPStatusError),  
    )  
    def _post(self, payload: dict) -> str:  
        """发一次请求；超时与 4xx/5xx 交给装饰器重试。"""  
        resp = httpx.post(  
            f"{self.base_url}/chat/completions",  
            headers={"Authorization": f"Bearer {self.api_key}"},  
            json=payload,  
            timeout=self.timeout,  
        )  
        resp.raise_for_status()  
        return resp.json()["choices"][0]["message"]["content"]  
  
    def chat(self, messages: list[ChatMessage]) -> str:  
        """发送消息并返回文本，失败抛 UpstreamError。"""  
        payload = {  
            "model": self.model,  
            "messages": [m.model_dump() for m in messages],  
        }  
        try:  
            return self._post(payload)  
        except httpx.HTTPError as exc:  
            logger.error("模型调用失败: %s", exc)  
            raise UpstreamError("模型服务不可用") from exc  
        except (KeyError, IndexError) as exc:  
            logger.error("模型返回结构异常: %s", exc)  
            raise UpstreamError("模型返回结构异常") from exc  
```

## 5. `app/agent/base.py`  
  
```python  
# 职责：定义所有工具必须实现的接口与自描述。  
# 实现步骤：Q-04  
# 被谁调用：app/agent/registry.py、app/agent/tools/kb_search.py  
# ── 【Q-04】工具基类 ──  
"""工具抽象接口。"""  
from abc import ABC, abstractmethod  
  
from app.agent.schemas import ToolSpec  
  
  
class BaseTool(ABC):  
    """所有工具的基类：名称、描述、参数、执行。"""  
  
    name: str = ""  
    description: str = ""  
    parameters: dict = {}  
  
    @abstractmethod  
    def run(self, **kwargs) -> str:  
        """执行工具，返回文本观察结果。"""  
  
    def spec(self) -> ToolSpec:  
        """生成给模型看的工具描述。"""  
        return ToolSpec(  
            name=self.name,  
            description=self.description,  
            parameters=self.parameters,  
        )  
```

## 6. `app/agent/registry.py`  
  
```python  
# 职责：保存工具实例，按名称调用，并把参数错误转成可读文本。  
# 实现步骤：Q-05  
# 被谁调用：app/agent/executor.py、scripts/ask.py、app/api/v1/chat.py  
# ── 【Q-05】工具注册表 ──  
"""工具注册与调度。"""  
from app.agent.base import BaseTool  
from app.agent.schemas import ToolSpec  
from app.core.logging import get_logger  
  
logger = get_logger(__name__)  
  
  
class ToolRegistry:  
    """名字到工具实例的映射。"""  
  
    def __init__(self):  
        """初始化空注册表。"""  
        self._tools: dict[str, BaseTool] = {}  
  
    def register(self, tool: BaseTool) -> None:  
        """注册一个工具，重名覆盖。"""  
        self._tools[tool.name] = tool  
  
    def specs(self) -> list[ToolSpec]:  
        """返回全部工具描述，供提示词拼接。"""  
        return [tool.spec() for tool in self._tools.values()]  
  
    def run(self, name: str, args: dict) -> str:  
        """按名调用；未知工具或参数错误返回可读文本。"""  
        tool = self._tools.get(name)  
        if tool is None:  
            return f"未知工具: {name}"  
        try:  
            return tool.run(**args)  
        except TypeError as exc:  
            logger.warning("工具参数错误 | tool=%s | %s", name, exc)  
            return f"参数错误: {exc}"  
```

## 7. `app/repositories/chunk_repo.py`  
  
```python  
# 职责：按关键词检索分块，并统计命中总数。  
# 实现步骤：Q-06  
# 被谁调用：app/services/search_service.py、tests/test_pipeline.py  
# ── 【Q-06】分块检索仓储 ──  
"""分块检索仓储。"""  
from sqlalchemy import func, select  
from sqlalchemy.orm import Session  
  
from app.models.chunk import Chunk  
  
  
def search_chunks(  
    db: Session,  
    keyword: str,  
    limit: int = 20,  
    offset: int = 0,  
) -> list[Chunk]:  
    """按关键词检索分块，按文档与块序号排序。"""  
    stmt = (  
        select(Chunk)  
        .where(Chunk.text.contains(keyword, autoescape=True))  
        .order_by(Chunk.document_id, Chunk.index)  
        .offset(offset)  
        .limit(limit)  
    )  
    return list(db.scalars(stmt).all())  
  
  
def count_chunks(db: Session, keyword: str) -> int:  
    """统计命中条数，用于分页总数。"""  
    stmt = (  
        select(func.count())  
        .select_from(Chunk)  
        .where(Chunk.text.contains(keyword, autoescape=True))  
    )  
    return db.scalar(stmt) or 0  
```

## 8. `app/services/search_service.py`  
  
```python  
# 职责：把页码换算成分页参数，返回当前页与总数。  
# 实现步骤：Q-07  
# 被谁调用：app/agent/tools/kb_search.py、tests/test_pipeline.py  
# ── 【Q-07】检索服务 ──  
"""分页检索。"""  
from sqlalchemy.orm import Session  
  
from app.repositories.chunk_repo import count_chunks, search_chunks  
  
  
def search_page(  
    db: Session,  
    keyword: str,  
    page: int = 1,  
    page_size: int = 20,  
) -> dict:  
    """返回当前页数据与总数，页码非法时归正为 1。"""  
    if page < 1:  
        page = 1  
    page_size = max(1, min(page_size, 50))  
    offset = (page - 1) * page_size  
  
    return {  
        "items": search_chunks(db, keyword, limit=page_size, offset=offset),  
        "total": count_chunks(db, keyword),  
        "page": page,  
        "page_size": page_size,  
    }  
```

## 10. `app/agent/tools/kb_search.py`  
  
```python  
# 职责：把检索服务包装成模型可调用的 kb_search 工具。  
# 实现步骤：Q-08  
# 被谁调用：scripts/ask.py、app/api/v1/chat.py、tests  
# ── 【Q-08】知识库检索工具 ──  
"""知识库检索工具。"""  
from sqlalchemy.orm import Session  
  
from app.agent.base import BaseTool  
from app.services.search_service import search_page  
  
  
class KBSearchTool(BaseTool):  
    """在知识库中按关键词检索分块。"""  
  
    name = "kb_search"  
    description = "在知识库中按关键词检索相关文本块，返回若干条片段"  
    parameters = {  
        "type": "object",  
        "properties": {"keyword": {"type": "string", "description": "检索关键词"}},  
        "required": ["keyword"],  
    }  
  
    def __init__(self, db: Session):  
        """注入数据库会话，供检索使用。"""  
        self.db = db  
  
    def run(self, keyword: str) -> str:  
        """执行检索，把前 5 块拼成文本。"""  
        page = search_page(self.db, keyword, page=1, page_size=5)  
        items = page["items"]  
        if not items:  
            return "未检索到相关内容"  
        return "\n".join(  
            f"[{i + 1}] {item.text}" for i, item in enumerate(items)  
        )  
```

## 12. `app/agent/executor.py`  
  
```python  
# 职责：组织消息、解析模型决策、调用工具，最多 5 轮。  
# 实现步骤：Q-10  
# 被谁调用：scripts/ask.py、app/api/v1/chat.py、tests  
# ── 【Q-10】ReAct 执行器 ──  
"""ReAct 执行器。"""  
from pydantic import ValidationError  
  
from app.agent.llm_client import LLMClient  
from app.agent.prompts import build_system_prompt  
from app.agent.registry import ToolRegistry  
from app.agent.schemas import AgentDecision, ChatMessage  
from app.core.logging import get_logger  
  
logger = get_logger(__name__)  
  
  
class AgentExecutor:  
    """按 ReAct 循环调用模型与工具。"""  
  
    def __init__(  
        self,  
        client: LLMClient,  
        registry: ToolRegistry,  
        max_steps: int = 5,  
    ):  
        """注入客户端、工具表与步数上限。"""  
        self.client = client  
        self.registry = registry  
        self.max_steps = max_steps  
  
    def run(self, question: str) -> str:  
        """回答问题，最多 max_steps 轮。"""  
        messages = [  
            ChatMessage(  
                role="system",  
                content=build_system_prompt(self.registry.specs()),  
            ),  
            ChatMessage(role="user", content=question),  
        ]  
  
        for step in range(1, self.max_steps + 1):  
            raw = self.client.chat(messages)  
            try:  
                decision = AgentDecision.model_validate_json(raw)  
            except ValidationError:  
                logger.warning("第 %d 轮输出非法，要求重答", step)  
                messages.append(ChatMessage(role="assistant", content=raw))  
                messages.append(  
                    ChatMessage(role="user", content="只输出合法 JSON。")  
                )  
                continue  
  
            if decision.action == "final":  
                logger.info(  
                    "第 %d 轮结束 | answer_len=%d", step, len(decision.answer or "")  
                )  
                return decision.answer or ""  
  
            try:  
                observation = self.registry.run(  
                    decision.tool or "", decision.args  
                )  
            except Exception as exc:  
                observation = f"工具执行失败: {exc}"  
  
            logger.info(  
                "第 %d 轮调用 %s | obs_len=%d", step, decision.tool, len(observation)  
            )  
            messages.append(ChatMessage(role="assistant", content=raw))  
            messages.append(  
                ChatMessage(  
                    role="user", content=f"观察结果: {observation[:500]}"  
                )  
            )  
  
        logger.warning("达到最大步数 | question=%s", question[:30])  
        return "达到最大步数，未能得出结论"  
```

## 13. `scripts/ask.py`  
  
```python  
# 职责：命令行问一句，直接走完整的在线链并打印答案。  
# 实现步骤：Q-11  
# 被谁调用：命令行 python -m scripts.ask "<问题>"  
# ── 【Q-11】命令行问答脚本 ──  
"""命令行问答。"""  
import sys  
  
from app.agent.executor import AgentExecutor  
from app.agent.llm_client import LLMClient  
from app.agent.registry import ToolRegistry  
from app.agent.tools.kb_search import KBSearchTool  
from app.core.logging import setup_logging  
from app.db.init_db import init_db  
from app.db.session import SessionLocal  
  
  
def main(question: str) -> None:  
    """初始化依赖、执行一轮问答、打印答案。"""  
    setup_logging()  
    init_db()  
  
    db = SessionLocal()  
    try:  
        registry = ToolRegistry()  
        registry.register(KBSearchTool(db))  
        answer = AgentExecutor(LLMClient(), registry).run(question)  
    finally:  
        db.close()  
  
    print(answer)  
  
  
if __name__ == "__main__":  
    if len(sys.argv) < 2:  
        print('用法: python -m scripts.ask "你的问题"')  
        raise SystemExit(1)  
    main(" ".join(sys.argv[1:]))  
```
