## 5. `app/api/v1/chat.py`  
  
```python  
# 职责：提供问答与流式问答两个接口。  
# 实现步骤：D-03, D-05  
# 被谁调用：app/main.py 挂载；浏览器与 curl  
# ── 【D-03】聊天路由与依赖注入 ──  
"""聊天接口。"""  
import time  
  
from fastapi import APIRouter, Depends  
from sqlalchemy.orm import Session  
  
from app.agent.executor import AgentExecutor  
from app.agent.llm_client import LLMClient  
from app.agent.registry import ToolRegistry  
from app.agent.tools.kb_search import KBSearchTool  
from app.api.v1.schemas import ChatRequest, ChatResponse  
from app.db.session import get_db  
  
router = APIRouter(prefix="/chat", tags=["chat"])  
  
_client = LLMClient()  
  
  
def get_executor(db: Session = Depends(get_db)) -> AgentExecutor:  
    """为每个请求构建执行器，工具绑定当前会话。"""  
    registry = ToolRegistry()  
    registry.register(KBSearchTool(db))  
    return AgentExecutor(_client, registry)  
  
  
@router.post("", response_model=ChatResponse, summary="问答")  
def chat(  
    req: ChatRequest,  
    executor: AgentExecutor = Depends(get_executor),  
) -> ChatResponse:  
    """接收问题，调用 Agent，返回答案与耗时。"""  
    start = time.perf_counter()  
    answer = executor.run(req.question)  
    elapsed = int((time.perf_counter() - start) * 1000)  
    return ChatResponse(answer=answer, elapsed_ms=elapsed)  
  
  
# ── 【D-05】流式路由 ──  
import json  
from collections.abc import Iterator  
  
from fastapi.responses import StreamingResponse  
  
  
def _to_sse(events: Iterator[dict]) -> Iterator[str]:  
    """把事件字典编码为 SSE 文本，并以 [DONE] 结束。"""  
    for event in events:  
        yield f"data: {json.dumps(event, ensure_ascii=False)}\n\n"  
    yield "data: [DONE]\n\n"  
  
  
@router.post("/stream", summary="流式问答")  
def chat_stream(  
    req: ChatRequest,  
    executor: AgentExecutor = Depends(get_executor),  
) -> StreamingResponse:  
    """按 SSE 逐段返回状态与答案。"""  
    return StreamingResponse(  
        _to_sse(executor.stream(req.question)),  
        media_type="text/event-stream",  
        headers={"Cache-Control": "no-cache"},  
    )  
```

## 6. `app/agent/executor.py`（替换第 2 批同名文件）  
  
```python  
# 职责：组织消息、解析决策、调用工具，并支持一次性与流式两种返回。  
# 实现步骤：Q-10, D-04  
# 被谁调用：scripts/ask.py、app/api/v1/chat.py、tests  
# ── 【Q-10】ReAct 执行器 ──  
"""ReAct 执行器。"""  
from collections.abc import Iterator  
  
from pydantic import ValidationError  
  
from app.agent.llm_client import LLMClient  
from app.agent.prompts import build_system_prompt  
from app.agent.registry import ToolRegistry  
from app.agent.schemas import AgentDecision, ChatMessage  
from app.core.exceptions import UpstreamError  
from app.core.logging import get_logger  
  
logger = get_logger(__name__)  
  
  
class AgentExecutor:  
    """按 ReAct 循环调用模型与工具。"""  
  
    def __init__(  
        self,  
        client: LLMClient,  
        registry: ToolRegistry,  
        max_steps: int = 5,  
    ):  
        """注入客户端、工具表与步数上限。"""  
        self.client = client  
        self.registry = registry  
        self.max_steps = max_steps  
  
    def _initial_messages(self, question: str) -> list[ChatMessage]:  
        """构造系统消息与用户问题。"""  
        return [  
            ChatMessage(  
                role="system",  
                content=build_system_prompt(self.registry.specs()),  
            ),  
            ChatMessage(role="user", content=question),  
        ]  
  
    def run(self, question: str) -> str:  
        """回答问题，最多 max_steps 轮。"""  
        messages = self._initial_messages(question)  
  
        for step in range(1, self.max_steps + 1):  
            raw = self.client.chat(messages)  
            try:  
                decision = AgentDecision.model_validate_json(raw)  
            except ValidationError:  
                logger.warning("第 %d 轮输出非法，要求重答", step)  
                messages.append(ChatMessage(role="assistant", content=raw))  
                messages.append(  
                    ChatMessage(role="user", content="只输出合法 JSON。")  
                )  
                continue  
  
            if decision.action == "final":  
                logger.info(  
                    "第 %d 轮结束 | answer_len=%d", step, len(decision.answer or "")  
                )  
                return decision.answer or ""  
  
            try:  
                observation = self.registry.run(  
                    decision.tool or "", decision.args  
                )  
            except Exception as exc:  
                observation = f"工具执行失败: {exc}"  
  
            logger.info(  
                "第 %d 轮调用 %s | obs_len=%d", step, decision.tool, len(observation)  
            )  
            messages.append(ChatMessage(role="assistant", content=raw))  
            messages.append(  
                ChatMessage(  
                    role="user", content=f"观察结果: {observation[:500]}"  
                )  
            )  
  
        logger.warning("达到最大步数 | question=%s", question[:30])  
        return "达到最大步数，未能得出结论"  
  
    # ── 【D-04】执行器流式方法 ──  
    def stream(self, question: str) -> Iterator[dict]:  
        """按轮次产出事件：status → tool → final／error。"""  
        messages = self._initial_messages(question)  
  
        try:  
            for step in range(1, self.max_steps + 1):  
                yield {"type": "status", "step": step, "message": "正在思考"}  
                raw = self.client.chat(messages)  
  
                try:  
                    decision = AgentDecision.model_validate_json(raw)  
                except ValidationError:  
                    messages.append(ChatMessage(role="assistant", content=raw))  
                    messages.append(  
                        ChatMessage(role="user", content="只输出合法 JSON。")  
                    )  
                    continue  
  
                if decision.action == "final":  
                    yield {"type": "final", "answer": decision.answer or ""}  
                    return  
  
                yield {"type": "tool", "step": step, "tool": decision.tool}  
  
                try:  
                    observation = self.registry.run(  
                        decision.tool or "", decision.args  
                    )  
                except Exception as exc:  
                    observation = f"工具执行失败: {exc}"  
  
                messages.append(ChatMessage(role="assistant", content=raw))  
                messages.append(  
                    ChatMessage(  
                        role="user", content=f"观察结果: {observation[:500]}"  
                    )  
                )  
  
            yield {"type": "final", "answer": "达到最大步数，未能得出结论"}  
        except UpstreamError as exc:  
            yield {"type": "error", "message": str(exc)}  
```

## 7. `app/core/exceptions.py`（替换第 1 批同名文件）  
  
```python  
# 职责：定义业务异常，并统一翻译成 HTTP 响应。  
# 实现步骤：O-05, D-07  
# 被谁调用：app/services/*、app/agent/*、app/main.py  
# ── 【O-05】定义业务异常 ──  
"""业务异常与全局处理器。"""  
  
  
class AppError(Exception):  
    """所有业务异常的基类。"""  
  
    status_code: int = 500  
    code: str = "INTERNAL_ERROR"  
    message: str = "服务器内部错误"  
  
    def __init__(self, message: str | None = None):  
        """允许调用方覆盖默认文案。"""  
        self.message = message or self.message  
        super().__init__(self.message)  
  
  
class NotFoundError(AppError):  
    """资源不存在。"""  
  
    status_code = 404  
    code = "NOT_FOUND"  
    message = "资源不存在"  
  
  
class InvalidInputError(AppError):  
    """业务参数不合法。"""  
  
    status_code = 400  
    code = "INVALID_INPUT"  
    message = "参数不合法"  
  
  
class UpstreamError(AppError):  
    """上游服务不可用。"""  
  
    status_code = 502  
    code = "UPSTREAM_ERROR"  
    message = "上游服务不可用"  
  
  
# ── 【D-07】注册全局异常处理器 ──  
from fastapi import FastAPI, Request  
from fastapi.exceptions import RequestValidationError  
from fastapi.responses import JSONResponse  
  
from app.core.logging import get_logger  
  
logger = get_logger(__name__)  
  
  
async def app_error_handler(request: Request, exc: AppError) -> JSONResponse:  
    """业务异常转统一响应。"""  
    logger.warning(  
        "业务异常 | path=%s | code=%s", request.url.path, exc.code  
    )  
    return JSONResponse(  
        status_code=exc.status_code,  
        content={"code": exc.code, "message": exc.message},  
    )  
  
  
async def validation_error_handler(  
    request: Request, exc: RequestValidationError  
) -> JSONResponse:  
    """字段校验失败转 422，结构与业务错误一致。"""  
    first = exc.errors()[0] if exc.errors() else {}  
    location = ".".join(str(part) for part in first.get("loc", []))  
    return JSONResponse(  
        status_code=422,  
        content={  
            "code": "VALIDATION_ERROR",  
            "message": f"参数校验失败: {location}",  
        },  
    )  
  
  
async def unhandled_error_handler(  
    request: Request, exc: Exception  
) -> JSONResponse:  
    """未知异常记堆栈，对外只给笼统信息。"""  
    logger.exception("未捕获异常 | path=%s", request.url.path)  
    return JSONResponse(  
        status_code=500,  
        content={"code": "INTERNAL_ERROR", "message": "服务器内部错误"},  
    )  
  
  
def register_exception_handlers(app: FastAPI) -> None:  
    """把三类处理器挂到应用。"""  
    app.add_exception_handler(AppError, app_error_handler)  
    app.add_exception_handler(RequestValidationError, validation_error_handler)  
    app.add_exception_handler(Exception, unhandled_error_handler)  
```
